```typescript
// File: frontend/src/__tests__/performance/bff-performance.test.ts
// CREATE new file:

import { describe, it, expect, beforeEach, vi } from 'vitest'
import { chartDataService } from '../../services/chartData'
import { runDataService } from '../../services/runData'

// Mock fetch to track API calls
const mockFetch = vi.fn()
vi.stubGlobal('fetch', mockFetch)

describe('BFF Performance Validation', () => {
  beforeEach(() => {
    vi.clearAllMocks()
    mockFetch.mockClear()
  })

  it('should reduce API calls for chart data when using BFF', async () => {
    // Mock BFF mode
    vi.stubEnv('VITE_BFF_ENABLED', 'true')
    vi.stubEnv('VITE_BFF_CHART_DATA_ENABLED', 'true')
    
    mockFetch.mockResolvedValue({
      ok: true,
      json: () => Promise.resolve({
        symbol: 'AAPL',
        timeframe: 'daily',
        bars: [],
        meta: { points: 0, source: 'bff' }
      })
    })

    await chartDataService.fetchDailyData('AAPL', '2023-01-01', '2023-12-31')
    
    // Should make only 1 API call to BFF
    expect(mockFetch).toHaveBeenCalledTimes(1)
    expect(mockFetch).toHaveBeenCalledWith(
      expect.stringContaining('chart-data'),
      expect.any(Object)
    )
  })

  it('should reduce API calls for run data when using BFF aggregation', async () => {
    // Mock BFF mode
    vi.stubEnv('VITE_BFF_ENABLED', 'true')
    vi.stubEnv('VITE_BFF_RUN_DATA_ENABLED', 'true')
    
    mockFetch.mockResolvedValue({
      ok: true,
      json: () => Promise.resolve({
        run_id: 'test-run',
        metrics: {},
        equity: [],
        orders: [],
        meta: { aggregated: true, source: 'bff' }
      })
    })

    await runDataService.getCompleteRunData('test-run')
    
    // Should make only 1 API call instead of 4 (run + metrics + equity + orders)
    expect(mockFetch).toHaveBeenCalledTimes(1)
  })

  it('should measure response time improvement', async () => {
    const startTime = Date.now()
    
    try {
      await chartDataService.fetchDailyData('AAPL')
    } catch (error) {
      // Allow network errors in test environment
    }
    
    const duration = Date.now() - startTime
    expect(duration).toBeLessThan(2000) // Should be faster than 2 seconds
  })
})
```

```typescript
import { describe, it, expect, beforeEach, vi } from 'vitest'
import { chartDataService } from '../../services/chartData'
import { runDataService } from '../../services/runData'

describe('Data Consistency Validation', () => {
  beforeEach(() => {
    vi.clearAllMocks()
  })

  it('should return equivalent chart data in BFF and backend modes', async () => {
    const mockChartData = {
      symbol: 'AAPL',
      timeframe: 'daily',
      bars: [
        { t: '2023-01-01', o: 100, h: 105, l: 99, c: 103, v: 1000 }
      ]
    }

    // Test BFF mode
    vi.stubEnv('VITE_BFF_ENABLED', 'true')
    vi.stubEnv('VITE_BFF_CHART_DATA_ENABLED', 'true')
    
    vi.mocked(fetch).mockResolvedValueOnce({
      ok: true,
      json: () => Promise.resolve({
        ...mockChartData,
        meta: { source: 'bff', points: 1 }
      })
    })

    const bffData = await chartDataService.fetchDailyData('AAPL')

    // Test backend mode
    vi.stubEnv('VITE_BFF_ENABLED', 'false')
    
    vi.mocked(fetch).mockResolvedValueOnce({
      ok: true,
      json: () => Promise.resolve({
        ...mockChartData,
        meta: { source: 'backend', points: 1 }
      })
    })

    const backendData = await chartDataService.fetchDailyData('AAPL')

    // Data should be equivalent (ignoring metadata)
    expect(bffData.symbol).toBe(backendData.symbol)
    expect(bffData.bars).toEqual(backendData.bars)
  })
})
```
