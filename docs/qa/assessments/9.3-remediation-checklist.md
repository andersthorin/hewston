```typescript
// File: frontend/src/__tests__/utils/websocket-test-harness.ts

export class WebSocketTestHarness {
  private mockWebSocket: MockWebSocket
  private messageQueue: any[] = []
  private connectionState: 'idle' | 'connecting' | 'connected' | 'closed' = 'idle'
  private eventHandlers: Map<string, Function[]> = new Map()

  constructor() {
    this.mockWebSocket = new MockWebSocket()
    this.setupMockWebSocket()
  }

  private setupMockWebSocket() {
    // Mock WebSocket constructor
    vi.stubGlobal('WebSocket', class MockWebSocket {
      static CONNECTING = 0
      static OPEN = 1
      static CLOSING = 2
      static CLOSED = 3

      readyState = MockWebSocket.CONNECTING
      url: string
      onopen: ((event: Event) => void) | null = null
      onmessage: ((event: MessageEvent) => void) | null = null
      onclose: ((event: CloseEvent) => void) | null = null
      onerror: ((event: Event) => void) | null = null

      constructor(url: string) {
        this.url = url
        // Simulate async connection
        setTimeout(() => {
          this.readyState = MockWebSocket.OPEN
          this.onopen?.(new Event('open'))
        }, 10)
      }

      send(data: string) {
        if (this.readyState === MockWebSocket.OPEN) {
          // Echo back for testing
          setTimeout(() => {
            this.onmessage?.(new MessageEvent('message', { data }))
          }, 5)
        }
      }

      close() {
        this.readyState = MockWebSocket.CLOSED
        setTimeout(() => {
          this.onclose?.(new CloseEvent('close'))
        }, 5)
      }
    })
  }

  async simulateConnection(url: string): Promise<void> {
    this.connectionState = 'connecting'
    // Simulate connection delay
    await new Promise(resolve => setTimeout(resolve, 10))
    this.connectionState = 'connected'
  }

  async simulateMessage(data: any): Promise<void> {
    if (this.connectionState === 'connected') {
      this.messageQueue.push(data)
      // Trigger message event
      const event = new MessageEvent('message', { data: JSON.stringify(data) })
      this.eventHandlers.get('message')?.forEach(handler => handler(event))
    }
  }

  async simulateDisconnection(): Promise<void> {
    this.connectionState = 'closed'
    const event = new CloseEvent('close')
    this.eventHandlers.get('close')?.forEach(handler => handler(event))
  }

  async simulateReconnection(): Promise<void> {
    await this.simulateConnection('mock-url')
  }

  getConnectionState() {
    return this.connectionState
  }

  getMessageQueue() {
    return [...this.messageQueue]
  }

  clearMessageQueue() {
    this.messageQueue = []
  }
}
```

```typescript
import { describe, it, expect, beforeEach } from 'vitest'
import { createWebSocketManager } from '../../services/websocket'
import { WebSocketTestHarness } from '../utils/websocket-test-harness'

describe('Connection Management Performance', () => {
  let testHarness: WebSocketTestHarness

  beforeEach(() => {
    testHarness = new WebSocketTestHarness()
  })

  it('should establish connection within 5 seconds', async () => {
    const wsManager = createWebSocketManager('test-run-123')
    const startTime = Date.now()
    
    await wsManager.connect()
    
    const connectionTime = Date.now() - startTime
    expect(connectionTime).toBeLessThan(5000) // 5 second max
    expect(wsManager.isReady()).toBe(true)
    
    wsManager.close()
  })

  it('should handle reconnection with exponential backoff', async () => {
    const wsManager = createWebSocketManager('test-run-123', {
      maxReconnectAttempts: 3,
      reconnectDelay: 100 // Fast for testing
    })
    
    await wsManager.connect()
    
    // Simulate multiple disconnections
    for (let i = 0; i < 3; i++) {
      await testHarness.simulateDisconnection()
      await new Promise(resolve => setTimeout(resolve, 150)) // Wait for reconnect attempt
    }
    
    const health = wsManager.getHealth()
    expect(health.reconnectAttempts).toBe(3)
    
    wsManager.close()
  })

  it('should queue messages during temporary disconnections', async () => {
    const wsManager = createWebSocketManager('test-run-123', {
      enableMessageQueue: true,
      maxQueueSize: 10
    })
    
    await wsManager.connect()
    
    // Simulate disconnection
    await testHarness.simulateDisconnection()
    
    // Send messages while disconnected
    wsManager.send('{"type": "play"}')
    wsManager.send('{"type": "pause"}')
    
    // Reconnect
    await testHarness.simulateReconnection()
    
    // Messages should be sent after reconnection
    const messageQueue = testHarness.getMessageQueue()
    expect(messageQueue.length).toBeGreaterThan(0)
    
    wsManager.close()
  })
})
```
