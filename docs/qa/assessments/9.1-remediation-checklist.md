import { describe, it, expect, beforeAll, afterAll } from 'vitest'
import { preview } from 'vite'
import type { PreviewServer } from 'vite'

describe('Vite Proxy Integration', () => {
  let server: PreviewServer

  beforeAll(async () => {
    server = await preview({
      preview: { port: 4173 }
    })
  })

  afterAll(async () => {
    await server.close()
  })

  it('should proxy BFF requests to port 8001', async () => {
    const response = await fetch('http://localhost:4173/api/v1/health')
    // Should attempt to proxy to BFF service
    expect(response).toBeDefined()
  })

  it('should proxy backend requests to port 8000', async () => {
    const response = await fetch('http://localhost:4173/backtests')
    // Should attempt to proxy to backend service
    expect(response).toBeDefined()
  })
})

#### 3.2 Create API Routing Integration Test
```typescript
// File: frontend/src/__tests__/integration/api-routing.test.ts
// CREATE new file:

import { describe, it, expect, beforeEach, vi } from 'vitest'
import { apiGetWithFlags } from '../../utils/api'

// Mock fetch for integration testing
global.fetch = vi.fn()

describe('API Routing Integration', () => {
  beforeEach(() => {
    vi.clearAllMocks()
    vi.resetModules()
  })

  it('should route to BFF when flags enabled', async () => {
    // Set environment for BFF mode
    vi.stubEnv('VITE_BFF_ENABLED', 'true')
    vi.stubEnv('VITE_BFF_CHART_DATA_ENABLED', 'true')
    
    vi.mocked(fetch).mockResolvedValue({
      ok: true,
      json: () => Promise.resolve({ data: 'bff-response' })
    } as Response)

    await apiGetWithFlags('/bars/daily', 'chartData')
    
    expect(fetch).toHaveBeenCalledWith(
      expect.stringContaining('127.0.0.1:8001'),
      expect.any(Object)
    )
  })

  it('should route to backend when flags disabled', async () => {
    // Set environment for backend mode
    vi.stubEnv('VITE_BFF_ENABLED', 'false')
    
    vi.mocked(fetch).mockResolvedValue({
      ok: true,
      json: () => Promise.resolve({ data: 'backend-response' })
    } as Response)

    await apiGetWithFlags('/bars/daily', 'chartData')
    
    expect(fetch).toHaveBeenCalledWith(
      expect.stringContaining('127.0.0.1:8000'),
      expect.any(Object)
    )
  })
})
```
